Fit linear model¶

model <- lm(Salary ~ YearsExperience, data=dataset)

Overlay the regression line¶

command = library(ggplot2) to enable graphing

ggplot() +

plot data points = geom_point(aes(x = dataset$YearsExperience, y = dataset$Salary), colour = 'red') +

plot regression line = geom_line(aes(x = dataset$YearsExperience, y = predict(model, newdata = dataset)), colour = 'blue') +

label graph title and x- and y-axis =

ggtitle('Salary vs Experience') +

xlab('Years of experience') +

ylab('Salary')
